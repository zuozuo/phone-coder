## 5. WebSocket Hub 设计

### 5.1 Hub 结构

```go
// internal/ws/hub.go
package ws

import (
    "sync"
)

type Hub struct {
    // 已连接的客户端
    clients map[*Client]bool

    // 客户端注册通道
    register chan *Client

    // 客户端注销通道
    unregister chan *Client

    // 广播消息
    broadcast chan []byte

    // 互斥锁
    mu sync.RWMutex
}

func NewHub() *Hub {
    return &Hub{
        clients:    make(map[*Client]bool),
        register:   make(chan *Client),
        unregister: make(chan *Client),
        broadcast:  make(chan []byte),
    }
}

func (h *Hub) Run() {
    for {
        select {
        case client := <-h.register:
            h.mu.Lock()
            h.clients[client] = true
            h.mu.Unlock()

        case client := <-h.unregister:
            h.mu.Lock()
            if _, ok := h.clients[client]; ok {
                delete(h.clients, client)
                close(client.send)
            }
            h.mu.Unlock()

        case message := <-h.broadcast:
            h.mu.RLock()
            for client := range h.clients {
                select {
                case client.send <- message:
                default:
                    close(client.send)
                    delete(h.clients, client)
                }
            }
            h.mu.RUnlock()
        }
    }
}
```

### 5.2 Client 结构

```go
// internal/ws/client.go
package ws

import (
    "encoding/json"
    "time"

    "github.com/gorilla/websocket"
)

const (
    writeWait      = 10 * time.Second
    pongWait       = 60 * time.Second
    pingPeriod     = (pongWait * 9) / 10
    maxMessageSize = 512 * 1024 // 512KB
)

type Client struct {
    hub  *Hub
    conn *websocket.Conn
    send chan []byte

    // 认证状态
    authenticated bool

    // 消息处理器
    handler MessageHandler
}

type MessageHandler interface {
    HandleMessage(client *Client, msg *ClientMessage) error
}

func (c *Client) ReadPump() {
    defer func() {
        c.hub.unregister <- c
        c.conn.Close()
    }()

    c.conn.SetReadLimit(maxMessageSize)
    c.conn.SetReadDeadline(time.Now().Add(pongWait))
    c.conn.SetPongHandler(func(string) error {
        c.conn.SetReadDeadline(time.Now().Add(pongWait))
        return nil
    })

    for {
        _, message, err := c.conn.ReadMessage()
        if err != nil {
            break
        }

        var msg ClientMessage
        if err := json.Unmarshal(message, &msg); err != nil {
            continue
        }

        if err := c.handler.HandleMessage(c, &msg); err != nil {
            // 记录错误日志
        }
    }
}

func (c *Client) WritePump() {
    ticker := time.NewTicker(pingPeriod)
    defer func() {
        ticker.Stop()
        c.conn.Close()
    }()

    for {
        select {
        case message, ok := <-c.send:
            c.conn.SetWriteDeadline(time.Now().Add(writeWait))
            if !ok {
                c.conn.WriteMessage(websocket.CloseMessage, []byte{})
                return
            }

            if err := c.conn.WriteMessage(websocket.TextMessage, message); err != nil {
                return
            }

        case <-ticker.C:
            c.conn.SetWriteDeadline(time.Now().Add(writeWait))
            if err := c.conn.WriteMessage(websocket.PingMessage, nil); err != nil {
                return
            }
        }
    }
}

func (c *Client) SendJSON(v interface{}) error {
    data, err := json.Marshal(v)
    if err != nil {
        return err
    }
    c.send <- data
    return nil
}
```

## 6. 消息路由处理

```go
// internal/handler/message_router.go
package handler

import (
    "phone-coder/internal/service"
    "phone-coder/internal/ws"
)

type MessageRouter struct {
    config         *config.Config
    sessionService *service.SessionService
    processService *service.ProcessService
}

func NewMessageRouter(cfg *config.Config, ss *service.SessionService, ps *service.ProcessService) *MessageRouter {
    return &MessageRouter{
        config:         cfg,
        sessionService: ss,
        processService: ps,
    }
}

func (r *MessageRouter) HandleMessage(client *ws.Client, msg *ws.ClientMessage) error {
    // 未认证时只允许 auth 消息
    if !client.Authenticated() && msg.Type != ws.MsgTypeAuth {
        return client.SendJSON(ws.ErrorMessage{
            Type:    ws.MsgTypeMessageError,
            Code:    "UNAUTHORIZED",
            Message: "请先认证",
        })
    }

    switch msg.Type {
    case ws.MsgTypeAuth:
        return r.handleAuth(client, msg)
    case ws.MsgTypeSessionList:
        return r.handleSessionList(client)
    case ws.MsgTypeSessionCreate:
        return r.handleSessionCreate(client, msg)
    case ws.MsgTypeSessionDelete:
        return r.handleSessionDelete(client, msg)
    case ws.MsgTypeMessageSend:
        return r.handleMessageSend(client, msg)
    case ws.MsgTypeSessionStop:
        return r.handleSessionStop(client, msg)
    default:
        return nil
    }
}

func (r *MessageRouter) handleAuth(client *ws.Client, msg *ws.ClientMessage) error {
    success := msg.Password == r.config.Server.Password

    if success {
        client.SetAuthenticated(true)
    }

    return client.SendJSON(ws.AuthResultMessage{
        Type:    ws.MsgTypeAuthResult,
        Success: success,
        Message: map[bool]string{true: "", false: "密码错误"}[success],
    })
}

func (r *MessageRouter) handleSessionList(client *ws.Client) error {
    sessions, err := r.sessionService.ListSessions()
    if err != nil {
        return err
    }

    return client.SendJSON(ws.SessionListMessage{
        Type:     ws.MsgTypeSessionListRes,
        Sessions: sessions,
        Workdirs: r.config.Workdirs,
    })
}

func (r *MessageRouter) handleSessionCreate(client *ws.Client, msg *ws.ClientMessage) error {
    // 验证目录是否在允许列表
    allowed := false
    for _, dir := range r.config.Workdirs {
        if dir == msg.WorkDir {
            allowed = true
            break
        }
    }
    if !allowed {
        return client.SendJSON(ws.ErrorMessage{
            Type:    ws.MsgTypeMessageError,
            Code:    ws.ErrCodeDirNotAllowed,
            Message: "该目录不在允许列表中",
        })
    }

    session, err := r.sessionService.CreateSession(msg.WorkDir)
    if err != nil {
        return err
    }

    return client.SendJSON(map[string]interface{}{
        "type":      ws.MsgTypeSessionCreated,
        "sessionId": session.ID,
        "name":      session.Name,
        "workDir":   session.WorkDir,
    })
}

func (r *MessageRouter) handleMessageSend(client *ws.Client, msg *ws.ClientMessage) error {
    session, err := r.sessionService.GetSession(msg.SessionID)
    if err != nil {
        return client.SendJSON(ws.ErrorMessage{
            Type:      ws.MsgTypeMessageError,
            SessionID: msg.SessionID,
            Code:      ws.ErrCodeSessionNotFound,
            Message:   "会话不存在",
        })
    }

    // 检查会话是否忙碌
    if session.Status == "running" {
        return client.SendJSON(ws.ErrorMessage{
            Type:      ws.MsgTypeMessageError,
            SessionID: msg.SessionID,
            MessageID: msg.MessageID,
            Code:      ws.ErrCodeCLIBusy,
            Message:   "请等待当前任务完成",
        })
    }

    // 发送确认
    client.SendJSON(map[string]interface{}{
        "type":      ws.MsgTypeMessageSent,
        "sessionId": msg.SessionID,
        "messageId": msg.MessageID,
    })

    // 更新会话状态
    r.sessionService.UpdateStatus(msg.SessionID, "running")

    // 发送消息到 Claude Code 进程
    go r.processService.SendMessage(msg.SessionID, msg.Content, func(content string) {
        // 流式输出回调
        client.SendJSON(ws.StreamMessage{
            Type:      ws.MsgTypeMessageStream,
            SessionID: msg.SessionID,
            Content:   content,
        })
    }, func() {
        // 完成回调
        r.sessionService.UpdateStatus(msg.SessionID, "idle")
        client.SendJSON(ws.DoneMessage{
            Type:      ws.MsgTypeMessageDone,
            SessionID: msg.SessionID,
        })
    })

    return nil
}

func (r *MessageRouter) handleSessionStop(client *ws.Client, msg *ws.ClientMessage) error {
    err := r.processService.StopProcess(msg.SessionID)
    if err != nil {
        return client.SendJSON(ws.ErrorMessage{
            Type:      ws.MsgTypeSessionStopErr,
            SessionID: msg.SessionID,
            Code:      ws.ErrCodeStopFailed,
            Message:   "停止失败",
        })
    }

    r.sessionService.UpdateStatus(msg.SessionID, "idle")

    return client.SendJSON(map[string]interface{}{
        "type":      ws.MsgTypeSessionStopped,
        "sessionId": msg.SessionID,
        "status":    "idle",
    })
}
```

## 7. Claude Code 进程管理

```go
// pkg/claude/process.go
package claude

import (
    "bufio"
    "io"
    "os"
    "os/exec"
    "sync"
    "syscall"
    "time"
)

type Process struct {
    cmd    *exec.Cmd
    stdin  io.WriteCloser
    stdout io.ReadCloser
    stderr io.ReadCloser

    mu      sync.Mutex
    running bool
}

func NewProcess(workDir string) (*Process, error) {
    cmd := exec.Command("claude")
    cmd.Dir = workDir
    cmd.Env = append(os.Environ(), "TERM=xterm-256color")

    stdin, err := cmd.StdinPipe()
    if err != nil {
        return nil, err
    }

    stdout, err := cmd.StdoutPipe()
    if err != nil {
        return nil, err
    }

    stderr, err := cmd.StderrPipe()
    if err != nil {
        return nil, err
    }

    if err := cmd.Start(); err != nil {
        return nil, err
    }

    return &Process{
        cmd:     cmd,
        stdin:   stdin,
        stdout:  stdout,
        stderr:  stderr,
        running: true,
    }, nil
}

func (p *Process) SendMessage(content string, onStream func(string), onDone func()) error {
    p.mu.Lock()
    defer p.mu.Unlock()

    if !p.running {
        return ErrProcessNotRunning
    }

    // 写入 stdin
    _, err := p.stdin.Write([]byte(content + "\n"))
    if err != nil {
        return err
    }

    // 读取 stdout
    go func() {
        scanner := bufio.NewScanner(p.stdout)
        for scanner.Scan() {
            line := scanner.Text()
            onStream(line + "\n")
        }
        onDone()
    }()

    return nil
}

func (p *Process) Stop() error {
    p.mu.Lock()
    defer p.mu.Unlock()

    if !p.running {
        return nil
    }

    // 先发送 SIGINT
    if err := p.cmd.Process.Signal(syscall.SIGINT); err != nil {
        return err
    }

    // 等待 3 秒
    done := make(chan error, 1)
    go func() {
        done <- p.cmd.Wait()
    }()

    select {
    case <-done:
        p.running = false
        return nil
    case <-time.After(3 * time.Second):
        // 超时，发送 SIGKILL
        p.cmd.Process.Kill()
        p.running = false
        return nil
    }
}

func (p *Process) IsRunning() bool {
    p.mu.Lock()
    defer p.mu.Unlock()
    return p.running
}

var ErrProcessNotRunning = errors.New("process not running")
```

## 10. SQLite 存储

```go
// internal/store/sqlite.go
package store

import (
    "gorm.io/driver/sqlite"
    "gorm.io/gorm"
    "phone-coder/internal/model"
)

type SQLiteStore struct {
    db *gorm.DB
}

func NewSQLiteStore(dbPath string) (*SQLiteStore, error) {
    db, err := gorm.Open(sqlite.Open(dbPath), &gorm.Config{})
    if err != nil {
        return nil, err
    }

    // 自动迁移
    db.AutoMigrate(&model.Session{})

    return &SQLiteStore{db: db}, nil
}

func (s *SQLiteStore) ListSessions() ([]model.Session, error) {
    var sessions []model.Session
    result := s.db.Order("updated_at DESC").Find(&sessions)
    return sessions, result.Error
}

func (s *SQLiteStore) GetSession(id string) (*model.Session, error) {
    var session model.Session
    result := s.db.First(&session, "id = ?", id)
    return &session, result.Error
}

func (s *SQLiteStore) CreateSession(session *model.Session) error {
    return s.db.Create(session).Error
}

func (s *SQLiteStore) DeleteSession(id string) error {
    return s.db.Delete(&model.Session{}, "id = ?", id).Error
}

func (s *SQLiteStore) UpdateSessionStatus(id, status string) error {
    return s.db.Model(&model.Session{}).Where("id = ?", id).Updates(map[string]interface{}{
        "status":     status,
        "updated_at": time.Now(),
    }).Error
}
```

```go
// internal/config/config.go
package config

import (
    "github.com/spf13/viper"
)

type Config struct {
    Server   ServerConfig `mapstructure:"server"`
    Workdirs []string     `mapstructure:"workdirs"`
}

type ServerConfig struct {
    Port     int    `mapstructure:"port"`
    Password string `mapstructure:"password"`
}

func Load(path string) (*Config, error) {
    viper.SetConfigFile(path)
    viper.SetConfigType("yaml")

    if err := viper.ReadInConfig(); err != nil {
        return nil, err
    }

    var cfg Config
    if err := viper.Unmarshal(&cfg); err != nil {
        return nil, err
    }

    return &cfg, nil
}
```

## 12. 程序入口

```go
// cmd/server/main.go
package main

import (
    "flag"
    "fmt"
    "log"
    "net/http"

    "github.com/gorilla/websocket"
    "phone-coder/internal/config"
    "phone-coder/internal/handler"
    "phone-coder/internal/service"
    "phone-coder/internal/store"
    "phone-coder/internal/ws"
)

var upgrader = websocket.Upgrader{
    ReadBufferSize:  1024,
    WriteBufferSize: 1024,
    CheckOrigin: func(r *http.Request) bool {
        return true // 允许所有来源（生产环境应限制）
    },
}

func main() {
    configPath := flag.String("config", "config.yaml", "配置文件路径")
    flag.Parse()

    // 加载配置
    cfg, err := config.Load(*configPath)
    if err != nil {
        log.Fatalf("加载配置失败: %v", err)
    }

    // 初始化存储
    store, err := store.NewSQLiteStore("phone-coder.db")
    if err != nil {
        log.Fatalf("初始化数据库失败: %v", err)
    }

    // 初始化服务
    sessionService := service.NewSessionService(store)
    processService := service.NewProcessService()

    // 初始化消息路由
    router := handler.NewMessageRouter(cfg, sessionService, processService)

    // 初始化 WebSocket Hub
    hub := ws.NewHub()
    go hub.Run()

    // WebSocket 处理
    http.HandleFunc("/ws", func(w http.ResponseWriter, r *http.Request) {
        conn, err := upgrader.Upgrade(w, r, nil)
        if err != nil {
            log.Printf("WebSocket 升级失败: %v", err)
            return
        }

        client := ws.NewClient(hub, conn, router)
        hub.Register(client)

        go client.WritePump()
        go client.ReadPump()
    })

    // 健康检查
    http.HandleFunc("/health", func(w http.ResponseWriter, r *http.Request) {
        w.WriteHeader(http.StatusOK)
        w.Write([]byte("OK"))
    })

    addr := fmt.Sprintf(":%d", cfg.Server.Port)
    log.Printf("服务启动在 %s", addr)
    log.Fatal(http.ListenAndServe(addr, nil))
}
```
