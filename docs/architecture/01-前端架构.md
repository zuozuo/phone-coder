## 4. WebSocket 消息协议

### 4.1 消息类型定义

```typescript
// types/ws.ts

// 客户端 -> 服务端
export type ClientMessage =
  | { type: 'auth'; password: string }
  | { type: 'session.list' }
  | { type: 'session.create'; workDir: string }
  | { type: 'session.delete'; sessionId: string }
  | { type: 'message.send'; sessionId: string; content: string; messageId: string }
  | { type: 'session.stop'; sessionId: string };

// 服务端 -> 客户端
export type ServerMessage =
  | { type: 'auth.result'; success: boolean; message?: string }
  | { type: 'session.list'; sessions: Session[]; workdirs: string[] }
  | { type: 'session.created'; sessionId: string; name: string; workDir: string }
  | { type: 'session.deleted'; sessionId: string }
  | { type: 'message.sent'; sessionId: string; messageId: string }
  | { type: 'message.stream'; sessionId: string; content: string }
  | { type: 'message.done'; sessionId: string }
  | { type: 'message.error'; sessionId: string; code: string; message: string }
  | { type: 'session.stopped'; sessionId: string; status: 'idle' }
  | { type: 'session.stop.error'; sessionId: string; code: string; message: string };
```

## 5. 状态管理 (Zustand)

### 5.1 authStore

```typescript
// stores/authStore.ts
import { create } from 'zustand';
import { persist } from 'zustand/middleware';

interface AuthState {
  isAuthenticated: boolean;
  serverUrl: string;

  setAuthenticated: (value: boolean) => void;
  setServerUrl: (url: string) => void;
  logout: () => void;
}

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      isAuthenticated: false,
      serverUrl: '',

      setAuthenticated: (value) => set({ isAuthenticated: value }),
      setServerUrl: (url) => set({ serverUrl: url }),
      logout: () => set({ isAuthenticated: false }),
    }),
    {
      name: 'auth-storage',
      partialize: (state) => ({ serverUrl: state.serverUrl }),
    }
  )
);
```

### 5.2 sessionStore

```typescript
// stores/sessionStore.ts
import { create } from 'zustand';

interface Session {
  id: string;
  name: string;
  workDir: string;
  status: 'running' | 'idle';
  updatedAt: string;
}

interface SessionState {
  sessions: Session[];
  workdirs: string[];
  currentSessionId: string | null;

  setSessions: (sessions: Session[]) => void;
  setWorkdirs: (workdirs: string[]) => void;
  setCurrentSession: (id: string | null) => void;
  updateSessionStatus: (id: string, status: 'running' | 'idle') => void;
  addSession: (session: Session) => void;
  removeSession: (id: string) => void;
}

export const useSessionStore = create<SessionState>((set) => ({
  sessions: [],
  workdirs: [],
  currentSessionId: null,

  setSessions: (sessions) => set({ sessions }),
  setWorkdirs: (workdirs) => set({ workdirs }),
  setCurrentSession: (id) => set({ currentSessionId: id }),
  updateSessionStatus: (id, status) => set((state) => ({
    sessions: state.sessions.map((s) =>
      s.id === id ? { ...s, status } : s
    ),
  })),
  addSession: (session) => set((state) => ({
    sessions: [session, ...state.sessions],
  })),
  removeSession: (id) => set((state) => ({
    sessions: state.sessions.filter((s) => s.id !== id),
  })),
}));
```

### 5.3 messageStore

```typescript
// stores/messageStore.ts
import { create } from 'zustand';

interface Message {
  id: string;
  sessionId: string;
  role: 'user' | 'assistant';
  content: string;
  status: 'sending' | 'sent' | 'streaming' | 'done' | 'error';
  createdAt: string;
}

interface MessageState {
  messages: Record<string, Message[]>;  // sessionId -> messages

  addMessage: (sessionId: string, message: Message) => void;
  updateMessageContent: (sessionId: string, messageId: string, content: string) => void;
  appendToLastMessage: (sessionId: string, content: string) => void;
  updateMessageStatus: (sessionId: string, messageId: string, status: Message['status']) => void;
  setLastMessageDone: (sessionId: string) => void;
}

export const useMessageStore = create<MessageState>((set) => ({
  messages: {},

  addMessage: (sessionId, message) => set((state) => ({
    messages: {
      ...state.messages,
      [sessionId]: [...(state.messages[sessionId] || []), message],
    },
  })),

  appendToLastMessage: (sessionId, content) => set((state) => {
    const msgs = state.messages[sessionId] || [];
    if (msgs.length === 0) return state;

    const lastMsg = msgs[msgs.length - 1];
    if (lastMsg.role !== 'assistant') {
      // 创建新的 assistant 消息
      return {
        messages: {
          ...state.messages,
          [sessionId]: [...msgs, {
            id: `assistant-${Date.now()}`,
            sessionId,
            role: 'assistant',
            content,
            status: 'streaming',
            createdAt: new Date().toISOString(),
          }],
        },
      };
    }

    // 追加到现有消息
    return {
      messages: {
        ...state.messages,
        [sessionId]: msgs.map((m, i) =>
          i === msgs.length - 1 ? { ...m, content: m.content + content } : m
        ),
      },
    };
  }),

  setLastMessageDone: (sessionId) => set((state) => {
    const msgs = state.messages[sessionId] || [];
    if (msgs.length === 0) return state;

    return {
      messages: {
        ...state.messages,
        [sessionId]: msgs.map((m, i) =>
          i === msgs.length - 1 ? { ...m, status: 'done' } : m
        ),
      },
    };
  }),

  updateMessageContent: (sessionId, messageId, content) => set((state) => ({
    messages: {
      ...state.messages,
      [sessionId]: (state.messages[sessionId] || []).map((m) =>
        m.id === messageId ? { ...m, content } : m
      ),
    },
  })),

  updateMessageStatus: (sessionId, messageId, status) => set((state) => ({
    messages: {
      ...state.messages,
      [sessionId]: (state.messages[sessionId] || []).map((m) =>
        m.id === messageId ? { ...m, status } : m
      ),
    },
  })),
}));
```

## 6. WebSocket 服务封装

```typescript
// services/wsService.ts
import { useAuthStore } from '../stores/authStore';
import { useSessionStore } from '../stores/sessionStore';
import { useMessageStore } from '../stores/messageStore';
import type { ClientMessage, ServerMessage } from '../types/ws';

class WebSocketService {
  private ws: WebSocket | null = null;
  private reconnectAttempts = 0;
  private maxReconnectAttempts = 5;

  connect(url: string): Promise<void> {
    return new Promise((resolve, reject) => {
      this.ws = new WebSocket(url);

      this.ws.onopen = () => {
        this.reconnectAttempts = 0;
        resolve();
      };

      this.ws.onmessage = (event) => {
        const message: ServerMessage = JSON.parse(event.data);
        this.handleMessage(message);
      };

      this.ws.onclose = () => {
        this.handleDisconnect();
      };

      this.ws.onerror = (error) => {
        reject(error);
      };
    });
  }

  private handleMessage(message: ServerMessage) {
    switch (message.type) {
      case 'auth.result':
        useAuthStore.getState().setAuthenticated(message.success);
        break;

      case 'session.list':
        useSessionStore.getState().setSessions(message.sessions);
        useSessionStore.getState().setWorkdirs(message.workdirs);
        break;

      case 'session.created':
        useSessionStore.getState().addSession({
          id: message.sessionId,
          name: message.name,
          workDir: message.workDir,
          status: 'idle',
          updatedAt: new Date().toISOString(),
        });
        break;

      case 'message.stream':
        useMessageStore.getState().appendToLastMessage(
          message.sessionId,
          message.content
        );
        useSessionStore.getState().updateSessionStatus(message.sessionId, 'running');
        break;

      case 'message.done':
        useMessageStore.getState().setLastMessageDone(message.sessionId);
        useSessionStore.getState().updateSessionStatus(message.sessionId, 'idle');
        break;

      case 'session.stopped':
        useSessionStore.getState().updateSessionStatus(message.sessionId, 'idle');
        break;

      // ... 其他消息处理
    }
  }

  private handleDisconnect() {
    useAuthStore.getState().setAuthenticated(false);

    if (this.reconnectAttempts < this.maxReconnectAttempts) {
      this.reconnectAttempts++;
      setTimeout(() => {
        const url = useAuthStore.getState().serverUrl;
        if (url) this.connect(url);
      }, 1000 * this.reconnectAttempts);
    }
  }

  send(message: ClientMessage) {
    if (this.ws?.readyState === WebSocket.OPEN) {
      this.ws.send(JSON.stringify(message));
    }
  }

  disconnect() {
    this.ws?.close();
    this.ws = null;
  }
}

export const wsService = new WebSocketService();
```

## 7. 自定义 Hooks

### 7.1 useWebSocket

```typescript
// hooks/useWebSocket.ts
import { useEffect, useCallback } from 'react';
import { wsService } from '../services/wsService';
import { useAuthStore } from '../stores/authStore';

export const useWebSocket = () => {
  const { serverUrl, isAuthenticated } = useAuthStore();

  const connect = useCallback(async (url: string, password: string) => {
    await wsService.connect(url);
    wsService.send({ type: 'auth', password });
  }, []);

  const disconnect = useCallback(() => {
    wsService.disconnect();
  }, []);

  return {
    connect,
    disconnect,
    isConnected: isAuthenticated,
  };
};
```

### 7.2 useAutoScroll

```typescript
// hooks/useAutoScroll.ts
import { useRef, useEffect, useCallback, useState } from 'react';

export const useAutoScroll = (dependency: unknown) => {
  const containerRef = useRef<HTMLDivElement>(null);
  const [isUserScrolling, setIsUserScrolling] = useState(false);

  const scrollToBottom = useCallback(() => {
    if (containerRef.current && !isUserScrolling) {
      containerRef.current.scrollTop = containerRef.current.scrollHeight;
    }
  }, [isUserScrolling]);

  useEffect(() => {
    scrollToBottom();
  }, [dependency, scrollToBottom]);

  const handleScroll = useCallback(() => {
    if (!containerRef.current) return;

    const { scrollTop, scrollHeight, clientHeight } = containerRef.current;
    const isAtBottom = scrollHeight - scrollTop - clientHeight < 100;
    setIsUserScrolling(!isAtBottom);
  }, []);

  return { containerRef, handleScroll, scrollToBottom };
};
```

```typescript
  ...rest
```

### 9.3 InputBar

```typescript
// components/business/InputBar.tsx
import { useState, useCallback } from 'react';
import { IonTextarea, IonButton, IonIcon } from '@ionic/react';
import { send, stop } from 'ionicons/icons';
import { wsService } from '../../services/wsService';
import { useSessionStore } from '../../stores/sessionStore';
import { useMessageStore } from '../../stores/messageStore';

interface Props {
  sessionId: string;
}

export const InputBar: React.FC<Props> = ({ sessionId }) => {
  const [text, setText] = useState('');
  const session = useSessionStore((s) => s.sessions.find((x) => x.id === sessionId));
  const isRunning = session?.status === 'running';

  const handleSend = useCallback(() => {
    if (!text.trim() || isRunning) return;

    const messageId = `msg-${Date.now()}`;

    // 添加用户消息到 store
    useMessageStore.getState().addMessage(sessionId, {
      id: messageId,
      sessionId,
      role: 'user',
      content: text,
      status: 'sending',
      createdAt: new Date().toISOString(),
    });

    // 发送 WebSocket 消息
    wsService.send({
      type: 'message.send',
      sessionId,
      content: text,
      messageId,
    });

    setText('');
  }, [text, sessionId, isRunning]);

  const handleStop = useCallback(() => {
    wsService.send({
      type: 'session.stop',
      sessionId,
    });
  }, [sessionId]);

  return (
    <div className="input-bar">
      <IonTextarea
        value={text}
        onIonChange={(e) => setText(e.detail.value || '')}
        placeholder="输入消息..."
        autoGrow
        disabled={isRunning}
      />
      {isRunning ? (
        <IonButton onClick={handleStop} color="danger">
          <IonIcon icon={stop} />
        </IonButton>
      ) : (
        <IonButton onClick={handleSend} disabled={!text.trim()}>
          <IonIcon icon={send} />
        </IonButton>
      )}
    </div>
  );
};
```

```typescript
import { Preferences } from '@capacitor/preferences';

export const secureStorage = {
  async set(key: string, value: string): Promise<void> {
    await Preferences.set({ key, value });
  },

  async get(key: string): Promise<string | null> {
    const { value } = await Preferences.get({ key });
```
